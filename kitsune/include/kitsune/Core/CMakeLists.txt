# The tablegen function below is what is used by LLVM and requires
# LLVM_TARGET_DEFINITIONS to be set to the source tablegen file.
set(LLVM_TARGET_DEFINITIONS LoopAttrs.td)

# The prefix LLVM here is used to construct the cmake variable LLVM_TABLEGEN_EXE
# which contains the path to the llvm-tblgen executable. At some point, we
# should probably set a corresponding KITSUNE_TABLEGEN_EXE variable and use
# that instead.
tablegen(LLVM LoopAttrs.inc -gen-kitsune-loop-attrs)

add_public_tablegen_target(kitsune_generated_loop_attrs)

# This generates the .inc files in ${KITSUNE_BINARY_DIR}/include. Here,
# ${KITSUNE_BINARY_DIR} is ${CMAKE_BINARY_DIR}/tools/kitsune since kitsune is
# treated by LLVM's build system as a tool (similar to clang)". LLVM does not
# look for header files in that directory. To avoid having to make changes to
# LLVM's build system, we copy the generated files to
# ${CMAKE_BINARY_DIR}/include since that directory is searched for included
# files.
add_custom_command(TARGET kitsune_generated_loop_attrs POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/include/kitsune/Core
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
  ${CMAKE_CURRENT_BINARY_DIR}/LoopAttrs.inc
  ${CMAKE_BINARY_DIR}/include/kitsune/Core/
  VERBATIM USES_TERMINAL)

# This is a "fake" target that is present to collect all .inc files that are
# generated by kitsune. Since some files in llvm/lib may depend on these,
# we can just provide this target as a dependency.
add_custom_target(kitsune_generated_attrs)
add_dependencies(kitsune_generated_attrs
  kitsune_generated_loop_attrs)
